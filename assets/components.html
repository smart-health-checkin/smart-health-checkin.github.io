    import { requestCheckin } from "smart-health-checkin";

// Ask the patient's wallet for two items.
export async function checkIn(verifier: Verifier, visitId: string) {
  const response = await requestCheckin(verifier, {
    purpose: "Before your visit",
    items: [{ id: "allergies", required: true }],
    timeoutMs: 120_000,
  });
  return response.requestStatus.filter((s) => s.status === "fulfilled");
}
